  \item 非線形連立方程式に対するニュートン法
    \[ {\bf x}^{(k+1)} = {\bf x}^{(k)} - J({\bf x}^{(k)})^{-1} f({\bf x}^{(k)}) \]

    \item 微分方程式の初期値問題の陰解法など

      \[ J({\bf x}^{(k)}){\bf \Delta x}=f({\bf x}^{(k)}) \qquad {\bf x}^{(k+1)}={\bf x}^{(k)}-{\bf \Delta x} \]
    \end{itemize}
